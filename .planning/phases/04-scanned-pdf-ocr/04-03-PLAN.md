```python
@router.get("/jobs/{job_id}/download")
async def download_artifact(
    job_id: str,
    artifact: str,  # bilingual_pdf | translated_pdf | translated_docx
    session: AsyncSession = Depends(get_session),
    settings: Settings = Depends(get_settings),
):
    """D-04-22: Download one of three compose-stage artifacts."""
    from fastapi.responses import FileResponse  # noqa: PLC0415

    job = await session.get(Job, job_id)
    if not job:
        raise HTTPException(status_code=404, detail="Job not found")
    if job.status not in (JobStatus.done, JobStatus.needs_review):
        raise HTTPException(status_code=409, detail="Job not yet complete")

    _artifact_map = {
        "bilingual_pdf":   ("output.pdf", "application/pdf"),
        "translated_pdf":  ("output-translated-only.pdf", "application/pdf"),
        "translated_docx": ("output.docx", "application/vnd.openxmlformats-officedocument.wordprocessingml.document"),
    }
    if artifact not in _artifact_map:
        raise HTTPException(status_code=400, detail=f"Unknown artifact '{artifact}'. Use: {', '.join(_artifact_map)}")

    filename, media_type = _artifact_map[artifact]
    file_path = os.path.join(settings.data_dir, "jobs", job_id, filename)
    if not os.path.exists(file_path):
        raise HTTPException(status_code=404, detail=f"Artifact '{artifact}' not found for this job")

    return FileResponse(
        path=file_path,
        media_type=media_type,
        filename=f"job-{job_id[:8]}-{filename}",
    )


@router.get("/jobs/{job_id}/pages/{page_n}.png")
async def serve_page_image(
    job_id: str,
    page_n: int,
    settings: Settings = Depends(get_settings),
):
    """Serve cached page PNG for image crop preview in review UI (D-04-10, D-04-11)."""
    from fastapi.responses import FileResponse  # noqa: PLC0415

    file_path = os.path.join(settings.data_dir, "jobs", job_id, "pages", f"page-{page_n}.png")
    if not os.path.exists(file_path):
        raise HTTPException(status_code=404, detail=f"Page {page_n} image not found")

    return FileResponse(path=file_path, media_type="image/png")
```
