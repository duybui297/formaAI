```python
import pymupdf
import os

def make_minimal_pdf(path: str, n_pages: int = 2) -> None:
    """Create an image-only PDF with blank pages for testing."""
    doc = pymupdf.open()
    for _ in range(n_pages):
        page = doc.new_page(width=595, height=842)  # A4 in points
        # Draw a simple rect to make it non-empty visually
        page.draw_rect(pymupdf.Rect(50, 50, 100, 100), color=(0, 0, 0), fill=(0.9, 0.9, 0.9))
    doc.save(path)
    doc.close()
```
