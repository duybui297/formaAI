      : pct >= 50

  return (
    <span
      className={cn("inline-block text-[10px] px-1.5 py-0.5 rounded border leading-tight", colorClass)}
      aria-label={`OCR confidence: ${pct}%`}
    >

    </span>
  )
}

const STAGE_COPY_TEMPLATE: Record<string, string> = {
  ocr: "OCR'd {current}/{total} pages",
  translate: "Translated {current}/{total} segments",
  compose: "Composing page {current}/{total}",
  reassemble: "Assembling outputs...",

function formatStageCopy(sp: { stage: string; current: number; total: number } | undefined): string {
  if (!sp) return ""
  const template = STAGE_COPY_TEMPLATE[sp.stage] ?? sp.stage

}

|----------|-------------|
| SegmentRow img src → /api/jobs/{id}/pages/{n}.png | Browser fetches page PNG; URL constructed from segment structural_position |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-14 | Information Disclosure | Confidence chip renders OCR confidence % | accept | Confidence is metadata about the user's own document; no sensitive data exposed |
| T-04-15 | Tampering | Source cell edited_source_text double-click | accept | User edits their own segment source; debounced PATCH validates max_length=10_000 server-side (SegmentPatchRequest in segments.py) |
| T-04-16 | Denial of Service | Image crop preview requests full page PNG | mitigate | Page PNGs are 300DPI files (500KB-2MB); browser cache deduplicates same-page requests across multiple segments; 25MB upload limit bounds page count |
| T-04-17 | Spoofing | isScannedOverride form field | accept | Server validates and re-runs detection logic; user overrides their own document classification only, no privilege escalation |
| T-04-18 | XSS | Image crop onError handler | mitigate | Error state uses React state (setImageError) + conditional render of static text paragraph — no DOM manipulation or dynamic HTML injection |
</threat_model>
